# 步骤3：认证令牌管理详细步骤
根据您提供的截图，前端已经完成了大部分配置，包括跨域设置和认证令牌管理。现在我们来详细说明认证令牌管理的具体步骤：

## 1. 令牌获取（登录接口）
前端需要调用后端的登录接口获取令牌：
// auth.ts
export async function login(username: string, password: string) {
  try {
    const response = await fetch('http://localhost:5186/api/Auth/login', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ username, password })
    });
    
    if (!response.ok) {
      throw new Error('登录失败');
    }
    
    const data = await response.json();
    // 存储令牌
    localStorage.setItem('token', data.accessToken);
    return data;
  } catch (error) {
    console.error('登录错误:', error);
    throw error;
  }
}
## 2. 令牌存储
使用localStorage存储令牌，确保页面刷新后仍然保持登录状态：
// 登录成功后存储令牌
localStorage.setItem('token', data.accessToken);

// 检查令牌是否存在
const token = localStorage.getItem('token');
if (token) {
  // 用户已登录
} else {
  // 用户未登录
}
## 3. 令牌使用（请求拦截器）
在每次请求中自动添加令牌到请求头：
// api/index.ts
import axios from 'axios';

const api = axios.create({
  baseURL: 'http://localhost:5186/api',
  headers: {
    'Content-Type': 'application/json'
  },
  withCredentials: true
});

// 请求拦截器 - 添加认证令牌
api.interceptors.request.use(config => {
  const token = localStorage.getItem('token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

export default api;
## 4. 令牌过期处理
处理令牌过期的情况，当收到401错误时跳转到登录页面：
// 响应拦截器 - 处理401错误
api.interceptors.response.use(
  response => response,
  error => {
    if (error.response && error.response.status === 401) {
      // 令牌过期，清除令牌并跳转到登录页面
      localStorage.removeItem('token');
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);
## 5. 令牌刷新
实现令牌刷新功能，当令牌即将过期时自动刷新：
// auth.ts
export async function refreshToken() {
  try {
    const response = await fetch('http://localhost:5186/api/Auth/refresh', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${localStorage.getItem('token')}`
      },
      body: JSON.stringify({
        refreshToken: localStorage.getItem('refreshToken')
      })
    });
    
    if (!response.ok) {
      throw new Error('刷新令牌失败');
    }
    
    const data = await response.json();
    // 更新令牌
    localStorage.setItem('token', data.accessToken);
    localStorage.setItem('refreshToken', data.refreshToken);
    return data;
  } catch (error) {
    console.error('刷新令牌错误:', error);
    // 刷新失败，跳转到登录页面
    localStorage.removeItem('token');
    localStorage.removeItem('refreshToken');
    window.location.href = '/login';
    throw error;
  }
}
## 6. 令牌清理（登出）
用户登出时清除令牌：
// auth.ts
export function logout() {
  // 清除令牌
  localStorage.removeItem('token');
  localStorage.removeItem('refreshToken');
  // 跳转到登录页面
  window.location.href = '/login';
}
